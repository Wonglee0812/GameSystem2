using DG.Tweening;
using TMPro;
using UnityEngine;

namespace PokerTD.UI
{
    /// <summary>
    /// Game information on two wooden signboards standing on the table (3D world UI):
    /// - top board, behind the map: ANTE (big title), WAVE under it, then a short hint for the phase;
    /// - right board, next to the map: sections ECONOMY, HEALTH, BUILD, HAND and COMBOS.
    /// The boards are built into the scene by the menu "Poker TD/Build New HUD In Scene", so they
    /// can be moved and restyled in the Editor. Without that, they are built at Play (fallback).
    /// They are not part of the battle board, so they stay while the board sinks into the shop.
    /// Numbers get a punch when they change.
    /// </summary>
    public class WorldHud : MonoBehaviour
    {
        [Header("Camera (set when the boards are built)")]
        public Vector3 cameraPosition = new Vector3(3.6f, 35f, -27f);
        public Vector3 cameraEuler = new Vector3(57f, 0f, 0f);

        [Header("Boards (used when building)")]
        public Vector3 topBoardPosition = new Vector3(0f, 0f, 11.2f);
        public Vector2 topBoardSize = new Vector2(12f, 3.6f);
        public Vector3 rightBoardPosition = new Vector3(18.6f, 0f, 0.2f);
        public Vector2 rightBoardSize = new Vector2(7.8f, 11.6f);

        [Header("Materials (assets; made by the builder)")]
        public Material woodDark;
        public Material woodDark2;
        public Material woodFrame;

        [Header("Texts (filled by the builder)")]
        public TextMeshPro ante;
        public TextMeshPro wave, hint, gold, lives, towers, machines, discards, deck, combos;

        public static readonly Color WoodDarkColor = new Color(0.36f, 0.21f, 0.12f);
        public static readonly Color WoodDark2Color = new Color(0.31f, 0.18f, 0.1f);
        public static readonly Color WoodFrameColor = new Color(0.78f, 0.45f, 0.25f);
        static readonly Color Text = new Color(0.98f, 0.93f, 0.82f);
        static readonly Color Header = new Color(1f, 0.78f, 0.28f);

        string lastAnte, lastWave, lastGold, lastLives, lastTowers, lastMachines, lastDiscards;

        void Start()
        {
            if (ante == null)   // not built into the scene: build now (fallback)
            {
                var baseMat = BuildManager.Instance != null ? BuildManager.Instance.material : null;
                if (woodDark == null) woodDark = Tinted(baseMat, WoodDarkColor);
                if (woodDark2 == null) woodDark2 = Tinted(baseMat, WoodDark2Color);
                if (woodFrame == null) woodFrame = Tinted(baseMat, WoodFrameColor);
                Build();
            }
            ComboView.WorldTarget = combos;
        }

        public static Material Tinted(Material source, Color color)
        {
            if (source == null) return null;
            var m = new Material(source);
            m.SetColor("_BaseColor", color);
            m.color = color;
            return m;
        }

        void OnDestroy()
        {
            if (ComboView.WorldTarget == combos) ComboView.WorldTarget = null;
        }

        /// <summary>Builds both boards as children of this object and moves the camera. Works in Edit mode.</summary>
        public void Build()
        {
            for (int i = transform.childCount - 1; i >= 0; i--) Kill(transform.GetChild(i).gameObject);
            var cam = Camera.main;
            if (cam != null)
            {
                cam.transform.position = cameraPosition;
                cam.transform.eulerAngles = cameraEuler;
            }

            // top board
            var top = Board("TopBoard", topBoardPosition, topBoardSize, 0.8f);
            ante = Label(top, "Ante", UiKit.Title, 11f, Header, new Vector2(0f, 0.9f), TextAlignmentOptions.Center, topBoardSize.x - 0.6f);
            wave = Label(top, "Wave", UiKit.Number, 9f, Text, new Vector2(0f, -0.25f), TextAlignmentOptions.Center, topBoardSize.x - 0.6f);
            hint = Label(top, "Hint", UiKit.Number, 5.5f, new Color(0.9f, 0.82f, 0.7f), new Vector2(0f, -1.2f), TextAlignmentOptions.Center, topBoardSize.x - 0.8f);
            ante.text = "ANTE 1"; wave.text = "WAVE 1 / 3"; hint.text = "PUT A CARD ON A TOWER TO START";

            // right board
            var right = Board("RightBoard", rightBoardPosition, rightBoardSize, 0.4f);
            float y = rightBoardSize.y * 0.5f - 0.6f;
            float w = rightBoardSize.x - 0.8f;
            Section(right, "ECONOMY", ref y, w);
            gold = Row(right, "tokens_stack", UiKit.Gold, ref y, w, "GOLD  15");
            Section(right, "HEALTH", ref y, w);
            lives = Row(right, "suit_hearts", UiKit.Red, ref y, w, "LIVES  20");
            Section(right, "BUILD", ref y, w);
            towers = Row(right, "structure_tower", Text, ref y, w, "TOWERS  0/5");
            machines = Row(right, "fire", new Color(0.6f, 0.85f, 1f), ref y, w, "MACHINES  0/5");
            Section(right, "HAND", ref y, w);
            discards = Row(right, "cards_fan", Text, ref y, w, "DISCARDS  3");
            deck = Row(right, "cards_stack", Text, ref y, w, "DECK  44");
            Section(right, "COMBOS", ref y, w);
            float h = y + rightBoardSize.y * 0.5f - 0.3f;   // room left down to the bottom edge
            combos = Label(right, "Combos", UiKit.Number, 3.4f, Text, new Vector2(0f, y + 0.35f - h * 0.5f), TextAlignmentOptions.TopLeft, w);
            combos.rectTransform.sizeDelta = new Vector2(w, h);
            combos.textWrappingMode = TextWrappingModes.Normal;   // the last line (best hand) is long
            combos.richText = true;
            combos.text = "COMBOS\nDrag cards onto towers.";
        }

        static void Kill(Object o)
        {
            if (Application.isPlaying) Destroy(o); else DestroyImmediate(o);
        }

        /// <summary>A sign: frame, planks, two posts. Returns the front plane (text goes on it, facing the camera).</summary>
        Transform Board(string name, Vector3 basePos, Vector2 size, float lift)
        {
            var board = new GameObject(name).transform;
            board.SetParent(transform, false);
            Vector3 centre = basePos + Vector3.up * (lift + size.y * 0.5f);
            board.position = centre;
            var cam = Camera.main;
            Vector3 toCam = cam != null ? cam.transform.position - centre : Vector3.back;
            board.rotation = Quaternion.LookRotation(-toCam.normalized, Vector3.up);   // +Z away from the camera

            Box(board, "Frame", new Vector3(0f, 0f, 0.06f), new Vector3(size.x + 0.45f, size.y + 0.45f, 0.22f), woodFrame);
            int planks = Mathf.Max(2, Mathf.RoundToInt(size.y / 0.9f));
            float ph = size.y / planks;
            for (int i = 0; i < planks; i++)
                Box(board, "Plank" + i, new Vector3(0f, -size.y * 0.5f + ph * (i + 0.5f), -0.06f), new Vector3(size.x, ph * 0.94f, 0.2f), i % 2 == 0 ? woodDark : woodDark2);
            for (int s = -1; s <= 1; s += 2)
            {
                Vector3 top = board.TransformPoint(new Vector3(s * (size.x * 0.5f - 0.5f), -size.y * 0.5f, 0.2f));
                Vector3 bottom = new Vector3(top.x, 0f, top.z);
                var post = Box(board, "Post", Vector3.zero, Vector3.one, woodFrame);
                post.position = (top + bottom) * 0.5f;
                post.rotation = Quaternion.identity;
                // world size 0.35 x height x 0.35, whatever the board's rotation
                post.SetParent(null, true);
                post.localScale = new Vector3(0.35f, Vector3.Distance(top, bottom) + 0.4f, 0.35f);
                post.SetParent(board, true);
            }
            var front = new GameObject("Front").transform;
            front.SetParent(board, false);
            front.localPosition = new Vector3(0f, 0f, -0.18f);
            return front;
        }

        static Transform Box(Transform parent, string name, Vector3 local, Vector3 scale, Material mat)
        {
            var go = GameObject.CreatePrimitive(PrimitiveType.Cube);
            Kill(go.GetComponent<Collider>());
            go.name = name;
            go.transform.SetParent(parent, false);
            go.transform.localPosition = local;
            go.transform.localScale = scale;
            if (mat != null) go.GetComponent<Renderer>().sharedMaterial = mat;
            return go.transform;
        }

        static TextMeshPro Label(Transform parent, string name, TMP_FontAsset font, float size, Color color, Vector2 pos, TextAlignmentOptions align, float width)
        {
            var t = new GameObject(name).AddComponent<TextMeshPro>();
            t.transform.SetParent(parent, false);
            t.rectTransform.sizeDelta = new Vector2(width, size * 0.12f);
            t.rectTransform.localPosition = new Vector3(pos.x, pos.y, 0f);
            t.rectTransform.localRotation = Quaternion.identity;   // the board's +Z points away from the camera
            UiKit.SetFont(t, font, 0.18f);
            t.fontSize = size;
            t.color = color;
            t.alignment = align;
            t.textWrappingMode = TextWrappingModes.NoWrap;
            return t;
        }

        static void Section(Transform board, string title, ref float y, float width)
        {
            y -= 0.12f;
            var t = Label(board, title, UiKit.Title, 3.6f, Header, new Vector2(0f, y), TextAlignmentOptions.Left, width);
            t.fontStyle = FontStyles.UpperCase;
            t.text = title;
            y -= 0.72f;
        }

        static TextMeshPro Row(Transform board, string icon, Color iconColor, ref float y, float width, string sample)
        {
            var sprite = UiKit.Icon(icon);
            if (sprite != null)
            {
                var go = new GameObject("Icon_" + icon);
                go.transform.SetParent(board, false);
                go.transform.localPosition = new Vector3(-width * 0.5f + 0.3f, y, 0f);
                go.transform.localScale = Vector3.one * 0.36f;
                var sr = go.AddComponent<SpriteRenderer>();
                sr.sprite = sprite;
                sr.color = iconColor;
            }
            var t = Label(board, "Row_" + icon, UiKit.Number, 5.2f, Text, new Vector2(0.45f, y), TextAlignmentOptions.Left, width - 0.9f);
            t.text = sample;
            y -= 0.6f;
            return t;
        }

        void Update()
        {
            var game = TDGame.Instance;
            if (game == null || ante == null) return;
            Set(ante, ref lastAnte, $"ANTE {game.Ante}");
            Set(wave, ref lastWave, $"WAVE {game.WaveInAnte} / {game.config.wavesPerAnte}");
            hint.text = Hint(game);
            Set(gold, ref lastGold, $"GOLD  {game.Gold}");
            Set(lives, ref lastLives, $"LIVES  {game.Lives}");
            Set(towers, ref lastTowers, $"TOWERS  {Tower.All.Count}/{game.TowerLimit}");
            Set(machines, ref lastMachines, $"MACHINES  {BuffMachine.All.Count}/{game.MachineLimit}{(game.MachineStock.Count > 0 ? $"  +{game.MachineStock.Count}" : "")}");
            Set(discards, ref lastDiscards, $"DISCARDS  {game.Hand.DiscardsLeft}");
            deck.text = $"DECK  {game.Hand.DeckCount}";
            bool fight = game.Current == TDGame.Phase.Deploy || game.Current == TDGame.Phase.Wave;
            if (!fight) combos.text = "<color=#BBA88F>Shown during the battle turn.</color>";
        }

        static string Hint(TDGame game)
        {
            if (game.PickingHandForItem >= 0) return "<color=#FFD860>PICK A HAND IN COMBOS (ESC CANCELS)</color>";
            switch (game.Current)
            {
                case TDGame.Phase.Deploy:
                    return game.CanStartWave ? "READY: PRESS START WAVE (SPACE)" : "<color=#FF9C7A>PUT A CARD ON A TOWER TO START</color>";
                case TDGame.Phase.Wave: return "WAVE IN PROGRESS";
                case TDGame.Phase.Shop: return "SHOP: BUY, THEN CONTINUE";
                case TDGame.Phase.GameOver: return "<color=#FF7060>GAME OVER: PRESS R</color>";
            }
            return "";
        }

        /// <summary>Sets the text; a changed value gets a punch (not on the first frame).</summary>
        static void Set(TextMeshPro t, ref string last, string value)
        {
            if (value == last) return;
            bool first = last == null;
            last = value;
            t.text = value;
            if (first) return;
            t.transform.DOComplete();
            t.transform.DOPunchScale(Vector3.one * 0.25f, 0.35f, 6, 0.8f).SetLink(t.gameObject);
        }
    }
}
